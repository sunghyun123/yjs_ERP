/**
 * 공사별·월별 성과/투입/손익 집계 — 매출손익 화면과 ERP MCP(get_profit)가 함께 쓴다.
 *
 * 이 파일이 손익 집계의 유일한 지점이다. 공식 자체(관리비 6%·준공 잔여)는 calc.ts와
 * junggong-seonggwa.ts에 있고, 여기서는 그것들을 불러 공사별·월별로 묶기만 한다.
 * 화면과 채팅이 같은 함수를 거쳐야 같은 "손익"이라는 단어가 같은 숫자를 가리킨다.
 *
 * 조회(load…)와 계산(aggregate…)을 나눈 이유: 화면은 세션 클라이언트(RLS),
 * MCP 라우트는 service_role 클라이언트를 넘긴다. 계산은 DB를 몰라야 테스트할 수 있다.
 */
import type { SupabaseClient } from '@supabase/supabase-js'
import type { Database, 공사단가Row } from '@/types/database'
import { fetchAllRows } from '@/lib/supabase/fetch-all'
import { calc합계, type 투입실적With상세 } from './calc'
import {
  build이력누계,
  calc준공잔여성과,
  load성과재료,
  sum준공잔여성과,
  type 공사이력Row,
  type 준공수주Row,
  type 준공잔여성과,
} from './junggong-seonggwa'
// 타입만 가져온다(빌드 후 사라짐) — 표 컴포넌트가 이 모양을 그대로 받는다
import type { PivotProjectRow } from '../sales/_components/PivotProjectTable'

export type 공사별손익재료 = {
  /** ⚠️ 선택 연도 것만 — load가 투입일로 잘라 온다. aggregate는 다시 자르지 않는다 */
  투입실적: 투입실적With상세[]
  단가목록: 공사단가Row[]
  /** 전 기간 — 준공 잔여 차감에 전 기간 누계가 필요하다. 월별 집계는 aggregate가 연도로 자른다 */
  공사이력전체: 공사이력Row[]
  수주: 준공수주Row[]
}

export type 공사별손익결과 = {
  pivotData: PivotProjectRow[]
  /** 월별 합계 (0 = 1월). 화면 KPI·차트와 MCP 합계가 이 배열에서 나온다 */
  monthly: { 성과: number; 투입: number }[]
  준공잔여목록: 준공잔여성과[]
  총준공반영액: number
}

function 연도범위(year: number) {
  return { yearStart: `${year}-01-01`, yearEnd: `${year + 1}-01-01` }
}

export async function load공사별손익재료(
  supabase: SupabaseClient<Database>,
  year: number,
): Promise<공사별손익재료> {
  const { yearStart, yearEnd } = 연도범위(year)
  const [투입실적행, 단가결과, 성과재료] = await Promise.all([
    fetchAllRows('투입실적', (from, to) =>
      supabase
        .from('투입실적')
        .select('*, 투입실적상세(투입구분, 주간수량, 야간수량)')
        .gte('투입일', yearStart)
        .lt('투입일', yearEnd)
        .order('id')
        .range(from, to),
    ),
    supabase.from('공사단가').select('*').order('적용시작일'),
    // 공사이력(전 기간) + 수주(준공 컬럼) — 홈 차트·홈 KPI도 같은 함수를 쓴다
    load성과재료(supabase),
  ])

  // 쿼리 실패 시 0/빈 데이터를 정상처럼 내보내지 않도록 명시적으로 에러를 던진다.
  // (fetchAllRows는 자기 안에서 던지므로 여기 남는 건 페이지네이션 안 하는 단가 하나)
  if (단가결과.error) throw new Error(`매출손익 데이터 조회 실패: ${단가결과.error.message}`)

  return {
    투입실적: 투입실적행 as unknown as 투입실적With상세[],
    단가목록: (단가결과.data ?? []) as 공사단가Row[],
    공사이력전체: 성과재료.공사이력,
    수주: 성과재료.수주,
  }
}

export function aggregate공사별손익(재료: 공사별손익재료, year: number): 공사별손익결과 {
  const { yearStart, yearEnd } = 연도범위(year)
  const { 투입실적: 투입실적목록, 단가목록, 공사이력전체, 수주: 수주목록 } = 재료

  // 월별 집계는 선택 연도만 — 공사이력은 성과누계용이라 전 기간이 들어온다
  const 공사이력목록 = 공사이력전체.filter(
    (r) => r.작업일자 >= yearStart && r.작업일자 < yearEnd,
  )

  type 수주Info = { 지중no: string; 공사명: string }
  const 수주Map = new Map<number, 수주Info>(
    수주목록.map((r) => [r.id, { 지중no: r.지중no, 공사명: r.공사명 }]),
  )

  // 준공 = 그 공사의 성과 총액이 준공액이 되도록 잔여분을 준공월에 얹는다 (파생, DB 무변경)
  const 준공잔여목록 = calc준공잔여성과(수주목록, build이력누계(공사이력전체), yearStart, yearEnd)
  const 준공잔여Map = new Map(준공잔여목록.map((r) => [r.수주_id, r]))
  const 총준공반영액 = sum준공잔여성과(준공잔여목록)

  // 월별(전체 차트용) + 공사별(월별) 단일 패스 집계
  const monthly = Array.from({ length: 12 }, () => ({ 성과: 0, 투입: 0 }))

  type 공사상세 = {
    monthly: { 성과: number; 투입: number }[]
  }
  const 공사별상세 = new Map<number, 공사상세>()

  function get공사상세(id: number): 공사상세 {
    if (!공사별상세.has(id)) {
      공사별상세.set(id, {
        monthly: Array.from({ length: 12 }, () => ({ 성과: 0, 투입: 0 })),
      })
    }
    return 공사별상세.get(id)!
  }

  for (const row of 투입실적목록) {
    const m = parseInt(row.투입일.slice(5, 7), 10) - 1
    const amt = calc합계(row, 단가목록)
    monthly[m].투입 += amt
    get공사상세(row.수주_id).monthly[m].투입 += amt
  }

  for (const row of 공사이력목록) {
    if (!row.작업일자) continue
    const m = parseInt(row.작업일자.slice(5, 7), 10) - 1
    const amt = row.성과금액 ?? 0
    monthly[m].성과 += amt
    get공사상세(row.수주_id).monthly[m].성과 += amt
  }

  // 준공 잔여성과를 준공월에 얹는다. 공사이력·투입실적이 한 줄도 없던 공사도 여기서 처음
  // 공사별상세에 등장한다(= 표에 새 행으로 나타난다) — 2026-07-31 요구사항의 의도다.
  for (const { 수주_id, 월, 금액 } of 준공잔여목록) {
    monthly[월].성과 += 금액
    get공사상세(수주_id).monthly[월].성과 += 금액
  }

  const pivotData: PivotProjectRow[] = [...공사별상세.entries()]
    .map(([id, { monthly: pm }]) => {
      const info = 수주Map.get(id)
      const 성과 = pm.reduce((s, m) => s + m.성과, 0)
      const 투입 = pm.reduce((s, m) => s + m.투입, 0)
      const 준공 = 준공잔여Map.get(id)
      return {
        id,
        지중no: info?.지중no ?? `(id:${id})`,
        공사명: info?.공사명 ?? '(공사명 없음)',
        성과금액: 성과,
        투입금액: 투입,
        손익금액: 성과 - 투입,
        monthly: pm.map(({ 성과, 투입 }) => ({ 성과, 투입, 손익: 성과 - 투입 })),
        // 성과에 이미 합쳐진 값 — 어느 달에 얼마가 준공으로 들어왔는지 구분하려고 같이 넘긴다
        준공: 준공 ? { 월: 준공.월, 금액: 준공.금액 } : undefined,
      }
    })
    .sort((a, b) => a.지중no.localeCompare(b.지중no, 'ko'))

 * ERP MCP 라우트(/api/mcp/*)의 키 검사·입력 해석·응답 모양 — DB를 모르는 순수 함수.

/**

export function mcp인증(
  authorization: string | null,
  apiKey: string | undefined,

export type 해석<T> = { ok: true; value: T } | { ok: false; error: string }
/** 서버가 partsKST()로 정한 지금 — 'current'는 이것으로만 해석한다 */
export type 지금 = { year: number; month: number }

const MIN_YEAR = 2000
const MAX_YEAR = 2100

function 연도해석(raw: string, 지금: 지금, 이름: string): 해석<number> {
  if (raw === 'current') return { ok: true, value: 지금.year }
  if (!/^\d{4}$/.test(raw)) {
    return { ok: false, error: `${이름}는 4자리 연도 또는 'current'여야 합니다: ${raw}` }
  }
  const y = Number(raw)
  if (y < MIN_YEAR || y > MAX_YEAR) {
    return { ok: false, error: `${이름}는 ${MIN_YEAR}~${MAX_YEAR} 사이여야 합니다: ${raw}` }
  }
  return { ok: true, value: y }
}

/**
 * year: 없으면 올해.
 * 매출손익 화면과 달리 잘못된 값을 올해로 바꾸지 않고 거절한다(→ 400).
 * "2025년"을 물었는데 2026년 숫자를 받아 2025년이라고 답하는 사고를 막는다.
 */
export function parse연도(raw: string | null, 지금: 지금): 해석<number> {
  if (raw === null || raw === '') return { ok: true, value: 지금.year }
  return 연도해석(raw, 지금, 'year')
}

/** month: 없으면 null(연간). 'current' 또는 1~12. */
export function parse월(raw: string | null, 지금: 지금): 해석<number | null> {
  if (raw === null || raw === '') return { ok: true, value: null }
  if (raw === 'current') return { ok: true, value: 지금.month }
  const m = /^\d{1,2}$/.test(raw) ? Number(raw) : NaN
  if (!(m >= 1 && m <= 12)) {
    return { ok: false, error: `month는 1~12 또는 'current'여야 합니다: ${raw}` }
  }
  return { ok: true, value: m }
}

/** completed_year: 없으면 null(준공 필터 없음). */
export function parse준공연도(raw: string | null, 지금: 지금): 해석<number | null> {
  if (raw === null || raw === '') return { ok: true, value: null }
  return 연도해석(raw, 지금, 'completed_year')
}

export type 손익행 = {
  지중no: string
  공사명: string
  성과: number
  투입: number
  손익: number

  준공반영액?: number

export type 손익응답 = {
  기준: { 연도: number; 월: number | null }
  단위: '원'
  합계: { 성과: number; 투입: number; 손익: number }
  공사: 손익행[]
  제외: { '성과·투입 모두 0인 공사': number }
}

export function build손익응답(결과: 공사별손익결과, 연도: number, 월: number | null): 손익응답 {
  const 월idx = 월 === null ? null : 월 - 1
  const 공사: 손익행[] = []
  let 제외 = 0

  for (const r of 결과.pivotData) {
    const 성과 = 월idx === null ? r.성과금액 : r.monthly[월idx].성과
    const 투입 = 월idx === null ? r.투입금액 : r.monthly[월idx].투입
    // 준공.월은 0-based. 월을 골랐으면 그 달에 얹힌 준공만 보인다
    const 준공반영액 =
      r.준공 && (월idx === null || r.준공.월 === 월idx) ? r.준공.금액 : undefined

    // 그 기간 움직임이 없는 공사는 0원 행만 늘린다 — 빼되 몇 건 뺐는지는 남긴다
    if (성과 === 0 && 투입 === 0 && 준공반영액 === undefined) {
      제외 += 1
      continue
    }

    const 행: 손익행 = {
      지중no: r.지중no,
      공사명: r.공사명,
      성과: Math.round(성과),
      투입: Math.round(투입),
      손익: Math.round(성과 - 투입),
    }
    if (준공반영액 !== undefined) 행.준공반영액 = Math.round(준공반영액)
    공사.push(행)
  }

  // 플러스 공사가 위, 마이너스 공사가 아래로 모인다
  공사.sort((a, b) => b.손익 - a.손익 || a.지중no.localeCompare(b.지중no, 'ko'))

  // 합계는 행 목록이 아니라 월별 합계에서 낸다 → 제외된 공사까지 포함하고,
  // 매출손익 화면 KPI·차트와 같은 원천이라 화면 숫자와 같아야 한다
  const 기간 = 월idx === null ? 결과.monthly : [결과.monthly[월idx]]
  const 성과합 = 기간.reduce((s, m) => s + m.성과, 0)
  const 투입합 = 기간.reduce((s, m) => s + m.투입, 0)

  return {
    기준: { 연도, 월 },
    단위: '원',
    합계: {
      성과: Math.round(성과합),
      투입: Math.round(투입합),
      손익: Math.round(성과합 - 투입합),
    },
    공사,
    제외: { '성과·투입 모두 0인 공사': 제외 },
  }
}

/**
 * '완결' = 준공여부 true (수주대장 '준공완료' 필터·도넛 '완료'와 같은 판정).
 * 연도는 준공일 문자열 앞 4자리로 본다 — new Date()를 쓰면 서버 시계(UTC)가 날짜를 민다.
 */
export function build공사목록응답(rows: 공사목록행[], 준공연도: number | null): 공사목록응답 {
  // 수주 전체가 0건이면 '공사가 없다'가 아니라 조회가 깨진 것 — 빈 목록을 정상처럼 내보내지 않는다
  if (rows.length === 0) throw new Error('수주 목록이 비어 있습니다. 조회가 정상이 아닙니다.')
  if (준공연도 === null) return { 기준: {}, 건수: rows.length, 공사: rows }
  const 접두 = `${준공연도}-`
  const 공사 = rows.filter((r) => r.준공여부 && r.준공일 !== null && r.준공일.startsWith(접두))
  return { 기준: { 준공연도 }, 건수: 공사.length, 공사 }
}

// 계산하지 않는다. 손익 공식은 ERP 서버(_lib/공사별손익) 한 곳에만 있고, 여기선 JSON을 받아 넘긴다.

export const REQUEST_TIMEOUT_MS = 30_000;
// 넘치면 Claude 쪽에서 응답 뒷부분이 조용히 잘릴 수 있다 → 자르지 않고 에러로 알린다
export const MAX_RESPONSE_CHARS = 150_000;

export class RelayError extends Error {}

export function parseServerUrl(raw) {
  let url;
  try {
    url = new URL(String(raw ?? "").trim());
  } catch {
    throw new RelayError("ERP 서버 주소 형식이 올바르지 않습니다. 확장 설정에서 주소를 확인해 주세요.");
  }
  const local = url.hostname === "127.0.0.1" || url.hostname === "localhost";
  // 키와 회사 금액이 오가는 통로라 https만 허용한다. 개발 PC의 로컬 서버 시험만 예외.
  if (url.protocol !== "https:" && !(local && url.protocol === "http:")) {
    throw new RelayError("ERP 서버 주소는 https:// 로 시작해야 합니다.");
  }
  url.pathname = url.pathname.replace(/\/+$/, "");
  url.search = "";
  url.hash = "";
  return url;
}

export function parseApiKey(raw) {
  const key = String(raw ?? "").trim();
  if (!key) {
    throw new RelayError("ERP 키가 설정되지 않았습니다. 확장 설정에서 키를 넣어 주세요.");
  }
  return key;
}

export function buildUrl(base, path, params = {}) {
  // 끝 슬래시를 떼고 붙인다. "//api/..."가 되면 URL이 'api'를 호스트 이름으로 읽는다.
  const prefix = base.pathname.replace(/\/+$/, "");
  const url = new URL(`${prefix}${path}`, base.origin);
  for (const [name, value] of Object.entries(params)) {
    if (value !== undefined && value !== null) url.searchParams.set(name, String(value));
  }
  return url;
}

// 상태코드로 '누구 쪽 문제인지'를 갈라 사용자가 할 수 있는 조치를 알려준다.
export function describeHttpFailure(status, serverMessage) {
  const detail = serverMessage ? ` — ${serverMessage}` : "";
  if (status === 400) return `요청 값이 잘못되었습니다(400)${detail}`;
  if (status === 401) {
    return "ERP가 키를 거절했습니다(401). 확장 설정의 키가 서버의 ERP_MCP_API_KEY와 같은지 확인해 주세요.";
  }
  if (status === 404) return "ERP에 조회 기능이 아직 배포되지 않았습니다(404). 관리자에게 알려 주세요.";
  if (status >= 500) return `ERP 서버에서 오류가 났습니다(${status})${detail}. 잠시 후 다시 시도해 주세요.`;
  return `요청이 거절되었습니다(${status})${detail}`;
}

export async function fetchJson(
  base,
  path,
  params,
  apiKey,
  { fetchImpl = fetch, timeoutMs = REQUEST_TIMEOUT_MS } = {},
) {
  const url = buildUrl(base, path, params);
  let response;
  try {
    response = await fetchImpl(url, {
      headers: { Accept: "application/json", Authorization: `Bearer ${apiKey}` },
      // 리다이렉트를 따라가면 Authorization 헤더(키)가 다른 서버로 실려 갈 수 있다
      redirect: "error",
      signal: AbortSignal.timeout(timeoutMs),
    });
  } catch (error) {
    if (error?.name === "TimeoutError") {
      throw new RelayError("ERP 조회가 시간 안에 끝나지 않았습니다. 잠시 후 다시 시도해 주세요.");
    }
    throw new RelayError("ERP 서버에 연결하지 못했습니다. 이 PC의 인터넷 연결을 확인해 주세요.");
  }
  if (!response.ok) {
    let message = "";
    try {
      message = (await response.json())?.error ?? "";
    } catch {
      // 본문이 JSON이 아니면 상태코드만으로 설명한다
    }
    throw new RelayError(describeHttpFailure(response.status, message));
  }
  return response.json();
}

export function toText(result, { maxChars = MAX_RESPONSE_CHARS } = {}) {
  const text = JSON.stringify(result, null, 1);
  if (text.length > maxChars) {
    throw new RelayError(
      `응답이 너무 큽니다(${text.length}자, 한도 ${maxChars}자). 월을 지정하거나 조건을 좁혀 다시 조회해 주세요.`,
    );
  }
  return text;
}

import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";

import {
  CONSTRUCTIONS_DESCRIPTION,
  PROFIT_DESCRIPTION,
  RelayError,
  fetchJson,
  parseApiKey,
  parseServerUrl,
  toText,
} from "./relay.js";

const server = new McpServer({ name: "yj-erp", version: "0.1.0" });

const 연도 = z.union([z.literal("current"), z.number().int().min(2000).max(2100)]);
const 월 = z.union([z.literal("current"), z.number().int().min(1).max(12)]);

async function 조회(도구, path, params) {
  try {
    // 설정은 호출마다 읽는다. 주소·키를 잘못 넣어도 서버 기동은 되고, 실패 이유가 대화에 그대로 보이게.
    const base = parseServerUrl(process.env.YJ_ERP_URL);
    const apiKey = parseApiKey(process.env.YJ_ERP_API_KEY);
    const result = await fetchJson(base, path, params, apiKey);
    return { content: [{ type: "text", text: toText(result) }] };
  } catch (error) {
    const text = error instanceof RelayError ? error.message : "ERP 조회에 실패했습니다.";
    // stdout은 MCP 통신 전용이라 진단은 stderr로만 남긴다(키·금액은 남기지 않는다).
    console.error(`[yj-erp] ${도구} 실패: ${error?.name ?? "Error"}`);
    return { isError: true, content: [{ type: "text", text }] };
  }
}

# yj-erp — ERP 조회 MCP 중계 (Claude Desktop 확장)

사장님 PC의 Claude Desktop에서 "올해 완결된 공사 몇 건?", "이번 달 손익 플러스/마이너스 나눠줘"를 쓰기 위한 중계.

|---|---|---|
| `get_constructions(completed_year?)` | `GET /api/mcp/constructions` | 지중no·공사명·준공여부·준공일. 준공 연도 필터 |

- [ ] **Step 2: 패키지 단독 실행 확인**

`.mcpb`는 zip이다. PowerShell `Expand-Archive`는 확장자가 `.zip`이어야 해서 복사본 이름을 바꿔 푼다 (레포 루트에서):

```powershell
Remove-Item -Recurse -Force tmp\mcpb-check -ErrorAction SilentlyContinue
New-Item -ItemType Directory -Force tmp\mcpb-check | Out-Null
Copy-Item mcp\dist\yj-erp.mcpb tmp\mcpb-check\pkg.zip
Expand-Archive tmp\mcpb-check\pkg.zip -DestinationPath tmp\mcpb-check\x
if (Test-Path tmp\mcpb-check\x\node_modules\@modelcontextprotocol\sdk\dist) { "sdk ok" }
if (-not (Test-Path tmp\mcpb-check\x\test)) { "no test dir ok" }
node -e "import('./tmp/mcpb-check/x/server/index.js').then(()=>{console.error('loaded');process.exit(0)})"
Remove-Item -Recurse -Force tmp\mcpb-check
```
Expected: `sdk ok`, `no test dir ok`, `loaded`.
